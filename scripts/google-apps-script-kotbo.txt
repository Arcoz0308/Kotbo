function onFormSubmit(e) {
  if (!e) {
    Logger.log("Aucun objet evenement fourni. Lancez cette fonction uniquement via un declencheur On form submit.");
    return;
  }

  var props = PropertiesService.getScriptProperties();
  var apiBaseUrl = props.getProperty("KOTBO_API_BASE_URL");
  var apiKey = props.getProperty("KOTBO_API_KEY");
  var bearerToken = props.getProperty("KOTBO_BEARER_TOKEN");
  var guildId = props.getProperty("KOTBO_GUILD_ID");

  if (!apiBaseUrl || (!apiKey && !bearerToken) || !guildId) {
    Logger.log("Config manquante: KOTBO_API_BASE_URL / KOTBO_GUILD_ID et (KOTBO_API_KEY ou KOTBO_BEARER_TOKEN)");
    return;
  }

  var endpoint = apiBaseUrl + "/api/dashboard/guilds/" + guildId + "/recruitment/candidatures";

  // Reponses du formulaire Google Forms
  var reponses = {};

  if (e.namedValues) {
    reponses = e.namedValues;
  } else if (e.response && typeof e.response.getItemResponses === "function") {
    var itemResponses = e.response.getItemResponses();
    for (var i = 0; i < itemResponses.length; i++) {
      var item = itemResponses[i].getItem();
      var itemTitle = item && typeof item.getTitle === "function" ? item.getTitle() : "Champ " + (i + 1);
      var responseValue = itemResponses[i].getResponse();
      reponses[itemTitle] = [String(responseValue ?? "")];
    }
  } else {
    Logger.log("Format d'evenement non supporte: impossible de lire les reponses du formulaire.");
    return;
  }

  // Optionnel: si le formulaire contient des champs explicites, on peut les extraire
  function firstAnswer(values, keys) {
    for (var i = 0; i < keys.length; i++) {
      var key = keys[i];
      if (values[key] && values[key][0]) return String(values[key][0]).trim();
    }
    return "";
  }

  function normalizeDiscordId(input) {
    if (!input) return "";
    var raw = String(input).trim();

    // Supporte les formats: 123..., <@123...>, <@!123...>
    var mention = raw.match(/^<@!?(\d{17,20})>$/);
    if (mention) return mention[1];

    var digits = raw.match(/\d{17,20}/);
    return digits ? digits[0] : "";
  }

  var discordIdRaw = firstAnswer(reponses, [
    "Discord ID",
    "ID Discord",
    "Quel est votre ID Discord ?",
    "Quel est votre ID Discord?"
  ]);
  var discordId = normalizeDiscordId(discordIdRaw);
  var username = (reponses["Pseudo"] && reponses["Pseudo"][0]) ? reponses["Pseudo"][0] : "";
  var email = (reponses["Email"] && reponses["Email"][0]) ? reponses["Email"][0] : "";

  if (!discordId) {
    Logger.log("ID Discord non detecte dans le formulaire (champ attendu: 'Quel est votre ID Discord ?').");
  }

  var submittedAt = new Date().toISOString();

  var payload = {
    data: reponses,
    submittedAt: submittedAt,
    discordId: discordId,
    username: username,
    email: email
  };

  var headers = {
    Accept: "application/json"
  };

  // Mode recommande: cle API Kotbo
  if (apiKey) {
    headers["X-Kotbo-Api-Key"] = apiKey;
    headers["X-Api-Key"] = apiKey;
  }

  // Mode legacy: JWT dashboard
  if (bearerToken) {
    headers.Authorization = "Bearer " + bearerToken;
  }

  var options = {
    method: "post",
    contentType: "application/json",
    headers: headers,
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  try {
    var response = UrlFetchApp.fetch(endpoint, options);
    var status = response.getResponseCode();
    var body = response.getContentText();

    if (status < 200 || status >= 300) {
      Logger.log("Diagnostic auth => hasApiKey=" + (!!apiKey) + " hasBearer=" + (!!bearerToken) + " guildId=" + guildId);
      Logger.log("Erreur Kotbo " + status + " : " + body);
      return;
    }

    Logger.log("Candidature Kotbo envoyee OK : " + body);
  } catch (err) {
    Logger.log("Exception envoi candidature Kotbo : " + err);
  }
}

  Logger.log("Cette fonction ne doit pas etre lancee manuellement sans objet evenement.");
}
